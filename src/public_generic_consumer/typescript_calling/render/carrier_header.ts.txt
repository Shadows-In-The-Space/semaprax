/** Generated flat-owned-Bytes reference codec; not the full compiler carrier.
 * All widths are exact u64 little-endian. Full validation precedes payload
 * allocation. Input records use own data fields, never getters or iteration.
 */
import { SemapraxPublicGenericException } from "./errors.js";
import type { CapacityReason, CarrierReason } from "./errors.js";
import type { Input, Output } from "./types.js";

export const MAX_OWNED_LEAVES = 256;
export const MAX_BYTES_PER_LEAF = 65536;
export const MAX_TOTAL_PAYLOAD_BYTES = 16777216;
export const MAX_CARRIER_BYTES = 8 + 8 * MAX_OWNED_LEAVES + MAX_TOTAL_PAYLOAD_BYTES;
const typedArrayPrototype: object = Object.getPrototypeOf(Uint8Array.prototype) as object;
const typedTag = Object.getOwnPropertyDescriptor(typedArrayPrototype, Symbol.toStringTag)!.get!;
const typedLength = Object.getOwnPropertyDescriptor(typedArrayPrototype, "length")!.get!;
const typedBuffer = Object.getOwnPropertyDescriptor(typedArrayPrototype, "buffer")!.get!;
const typedOffset = Object.getOwnPropertyDescriptor(typedArrayPrototype, "byteOffset")!.get!;
const arrayBufferLength = Object.getOwnPropertyDescriptor(ArrayBuffer.prototype, "byteLength")!.get!;
const arrayBufferResizable = Object.getOwnPropertyDescriptor(ArrayBuffer.prototype, "resizable")?.get;

function capacityExceeded(reason: CapacityReason): SemapraxPublicGenericException {
  return new SemapraxPublicGenericException({ kind: "capacity-exceeded", reason });
}
function resultRejected(reason: CarrierReason): SemapraxPublicGenericException {
  return new SemapraxPublicGenericException({ kind: "result-rejected", reason });
}
function inputRejected(reason: CarrierReason): SemapraxPublicGenericException {
  return new SemapraxPublicGenericException({ kind: "carrier-rejected", reason });
}

/** Intrinsic brand checks ignore forged .length/.buffer and Symbol.toStringTag.
 * Shared, resizable and detached buffers are outside this synchronous profile.
 * A genuine Uint8Array view is returned; user slice/iterator methods never run.
 */
export function byteView(value: unknown): Uint8Array {
  try {
    if (Reflect.apply(typedTag, value, []) !== "Uint8Array") throw inputRejected("input-bytes");
    const buffer: unknown = Reflect.apply(typedBuffer, value, []);
    Reflect.apply(arrayBufferLength, buffer, []); // Reject SharedArrayBuffer.
    if (arrayBufferResizable && Reflect.apply(arrayBufferResizable, buffer, [])) throw inputRejected("input-bytes");
    const offset: number = Reflect.apply(typedOffset, value, []);
    const length: number = Reflect.apply(typedLength, value, []);
    return new Uint8Array(buffer as ArrayBuffer, offset, length);
  } catch {
    throw inputRejected("input-bytes");
  }
}

function validCount(count: number): void {
  if (!Number.isInteger(count) || count < 1 || count > MAX_OWNED_LEAVES) throw capacityExceeded("leaf-count");
}

function readInputFields(value: unknown, names: readonly string[]): readonly Uint8Array[] {
  if (value === null || typeof value !== "object") throw inputRejected("input-shape");
  let proto: unknown;
  let keys: readonly PropertyKey[];
  // Host proxy traps are not entitled to forge a semantic exception. Catch
  // only reflective operations here; our own closed refusals happen outside.
  try { proto = Object.getPrototypeOf(value); keys = Reflect.ownKeys(value); }
  catch { throw inputRejected("input-shape"); }
  if (proto !== null && proto !== Object.prototype) throw inputRejected("input-shape");
  if (keys.length !== names.length || keys.some(key => typeof key !== "string" || !names.includes(key))) {
    throw inputRejected("input-field");
  }
  const leaves: Uint8Array[] = [];
  for (const name of names) {
    let property: PropertyDescriptor | undefined;
    try { property = Object.getOwnPropertyDescriptor(value, name); }
    catch { throw inputRejected("input-shape"); }
    if (!property || !("value" in property)) throw inputRejected("input-field");
    leaves.push(byteView(property.value));
  }
  return leaves;
}

export interface LeafSpan { readonly offset: number; readonly length: number; }

export function encodeLeaves(leaves: readonly Uint8Array[]): Uint8Array {
  if (!Array.isArray(leaves)) throw inputRejected("input-shape");
  validCount(leaves.length);
  let payload = 0;
  const views: Uint8Array[] = [];
  for (let i = 0; i < leaves.length; i += 1) {
    const property = Object.getOwnPropertyDescriptor(leaves, String(i));
    if (!property || !("value" in property)) throw inputRejected("input-field");
    const view = byteView(property.value);
    if (view.length > MAX_BYTES_PER_LEAF) throw capacityExceeded("leaf-bytes");
    if (view.length > MAX_TOTAL_PAYLOAD_BYTES - payload) throw capacityExceeded("total-payload");
    payload += view.length;
    views.push(view);
  }
  let out: Uint8Array;
  try { out = new Uint8Array(8 + 8 * views.length + payload); }
  catch { throw capacityExceeded("host-allocation"); }
  const wire = new DataView(out.buffer);
  wire.setBigUint64(0, BigInt(views.length), true);
  let offset = 8;
  for (const view of views) {
    wire.setBigUint64(offset, BigInt(view.length), true);
    offset += 8;
    out.set(view, offset);
    offset += view.length;
  }
  return out;
}

/** Returns spans only: even a hostile last leaf is checked before copying the
 * first payload. Copy code must not allocate result leaves during this pass. */
export function locateLeaves(bytes: Uint8Array, expectedCount: number): LeafSpan[] {
  validCount(expectedCount);
  let checked: Uint8Array;
  try { checked = byteView(bytes); } catch { throw resultRejected("carrier-framing"); }
  if (checked.length > MAX_CARRIER_BYTES) throw resultRejected("carrier-bytes");
  if (checked.length < 8) throw resultRejected("carrier-framing");
  const wire = new DataView(checked.buffer, checked.byteOffset, checked.byteLength);
  if (wire.getBigUint64(0, true) !== BigInt(expectedCount)) throw resultRejected("carrier-count");
  let offset = 8;
  let payload = 0;
  const spans: LeafSpan[] = [];
  for (let index = 0; index < expectedCount; index += 1) {
    if (checked.length - offset < 8) throw resultRejected("carrier-framing");
    const length = wire.getBigUint64(offset, true);
    offset += 8;
    if (length > BigInt(MAX_BYTES_PER_LEAF)) throw resultRejected("leaf-bytes");
    const exact = Number(length);
    if (exact > MAX_TOTAL_PAYLOAD_BYTES - payload) throw resultRejected("total-payload");
    if (exact > checked.length - offset) throw resultRejected("carrier-framing");
    payload += exact;
    spans.push({ offset, length: exact });
    offset += exact;
  }
  if (offset !== checked.length) throw resultRejected("carrier-trailing");
  return spans;
}

export function decodeLeaves(bytes: Uint8Array, expectedCount: number): Uint8Array[] {
  const checked = byteView(bytes);
  const spans = locateLeaves(checked, expectedCount);
  try { return spans.map(span => checked.slice(span.offset, span.offset + span.length)); }
  catch { throw capacityExceeded("host-allocation"); }
}

// Canonical LogicalCarrierFrame v1. The legacy helpers above remain solely
// for the explicitly selected reference provider; compiled providers use the
// descriptor-bound framing below. This implementation deliberately keeps the
// SHA-256 primitive synchronous: transform() is a synchronous owned call and
// Web Crypto's Promise API would otherwise create a second lifecycle shape.
const CARRIER_SCHEMA = "semaprax.public-generic-carrier.v1";
const FRAME_DOMAIN = new TextEncoder().encode("semaprax.public-generic-carrier.v1.frame\0");
const ENDPOINT_DOMAIN = new TextEncoder().encode("semaprax.public-generic-carrier.v1.endpoint-identity\0");
const INVENTORY_DOMAIN = new TextEncoder().encode("semaprax.public-generic-carrier.v1.leaf-inventory\0");
const MAX_FRAME_FIELD_BYTES = 65536;

function digest(domain: Uint8Array, body: Uint8Array): string { const preimage = new Uint8Array(domain.length + 8 + body.length); preimage.set(domain); new DataView(preimage.buffer).setBigUint64(domain.length, BigInt(body.length), true); preimage.set(body, domain.length + 8); return `sha256:${Array.from(sha256(preimage), byte => byte.toString(16).padStart(2, "0")).join("")}`; }
function framed(bytes: Uint8Array): Uint8Array { const out = new Uint8Array(8 + bytes.length); new DataView(out.buffer).setBigUint64(0, BigInt(bytes.length), true); out.set(bytes, 8); return out; }
function join(parts: readonly Uint8Array[]): Uint8Array { const length = parts.reduce((total, part) => total + part.length, 0); const out = new Uint8Array(length); let offset = 0; for (const part of parts) { out.set(part, offset); offset += part.length; } return out; }
function text(value: string): Uint8Array { return new TextEncoder().encode(value); }
function inventoryDigest(paths: readonly string[]): string { return digest(INVENTORY_DOMAIN, join([u64(paths.length), ...paths.map(path => framed(text(path)))])); }
function endpointDigest(exportId: string): string { return digest(ENDPOINT_DOMAIN, text(exportId)); }
function u64(value: number): Uint8Array { const out = new Uint8Array(8); new DataView(out.buffer).setBigUint64(0, BigInt(value), true); return out; }

export function encodeCanonicalFrame(direction: "input" | "result", descriptorDigest: string, exportId: string, instanceDigest: string, paths: readonly string[], leaves: readonly Uint8Array[]): Uint8Array {
  validCount(paths.length); if (leaves.length !== paths.length) throw inputRejected("carrier-count");
  let total = 0; const payloads: Uint8Array[] = [];
  for (const leaf of leaves) { const view = byteView(leaf); if (view.length > MAX_BYTES_PER_LEAF) throw capacityExceeded("leaf-bytes"); if (view.length > MAX_TOTAL_PAYLOAD_BYTES - total) throw capacityExceeded("total-payload"); total += view.length; payloads.push(view); }
  const prefix: Uint8Array[] = [framed(text(CARRIER_SCHEMA)), framed(text(direction)), framed(text(descriptorDigest)), framed(text(endpointDigest(exportId))), framed(text(instanceDigest)), framed(text(inventoryDigest(paths))), u64(paths.length), u64(total)];
  for (let index = 0; index < paths.length; index += 1) prefix.push(framed(text(paths[index]!)), Uint8Array.of(0), framed(payloads[index]!));
  const body = join(prefix); return join([body, framed(text(digest(FRAME_DOMAIN, body)))]);
}
function readFrame(bytes: Uint8Array, offset: number): readonly [Uint8Array, number] | null { if (bytes.length - offset < 8) return null; const width = new DataView(bytes.buffer, bytes.byteOffset + offset, 8).getBigUint64(0, true); if (width > BigInt(MAX_FRAME_FIELD_BYTES)) return null; const length = Number(width); const start = offset + 8; return length <= bytes.length - start ? [bytes.subarray(start, start + length), start + length] : null; }
function readText(bytes: Uint8Array, offset: number): readonly [string, number] | null { const found = readFrame(bytes, offset); if (!found) return null; try { return [new TextDecoder("utf-8", { fatal: true, ignoreBOM: true }).decode(found[0]), found[1]]; } catch { return null; } }
export function decodeCanonicalFrame(direction: "input" | "result", descriptorDigest: string, exportId: string, instanceDigest: string, paths: readonly string[], bytes: Uint8Array): Uint8Array[] {
  let checked: Uint8Array; try { checked = byteView(bytes); } catch { throw resultRejected("carrier-framing"); }
  if (checked.length > MAX_CARRIER_BYTES + 4 * 1024 * 1024) throw resultRejected("carrier-bytes");
  let offset = 0; const bodyStart = offset; const expected = [CARRIER_SCHEMA, direction, descriptorDigest, endpointDigest(exportId), instanceDigest, inventoryDigest(paths)];
  for (const value of expected) { const field = readText(checked, offset); if (!field || field[0] !== value) throw resultRejected("carrier-framing"); offset = field[1]; }
  if (checked.length - offset < 16) throw resultRejected("carrier-framing"); const view = new DataView(checked.buffer, checked.byteOffset, checked.byteLength); const count = view.getBigUint64(offset, true); offset += 8; const declaredTotal = view.getBigUint64(offset, true); offset += 8;
  if (count !== BigInt(paths.length)) throw resultRejected("carrier-count"); if (declaredTotal > BigInt(MAX_TOTAL_PAYLOAD_BYTES)) throw resultRejected("total-payload");
  let total = 0; const payloads: Uint8Array[] = [];
  for (let index = 0; index < paths.length; index += 1) { const path = readText(checked, offset); if (!path || path[0] !== paths[index]) throw resultRejected("carrier-framing"); offset = path[1]; if (offset >= checked.length || checked[offset] !== 0) throw resultRejected("carrier-framing"); offset += 1; const payload = readFrame(checked, offset); if (!payload || payload[0].length > MAX_BYTES_PER_LEAF) throw resultRejected("leaf-bytes"); offset = payload[1]; if (payload[0].length > MAX_TOTAL_PAYLOAD_BYTES - total) throw resultRejected("total-payload"); total += payload[0].length; payloads.push(payload[0]); }
  if (BigInt(total) !== declaredTotal) throw resultRejected("carrier-framing"); const body = checked.subarray(bodyStart, offset); const facts = readText(checked, offset); if (!facts || facts[1] !== checked.length || facts[0] !== digest(FRAME_DOMAIN, body)) throw resultRejected("carrier-framing");
  try { return payloads.map(payload => payload.slice()); } catch { throw capacityExceeded("host-allocation"); }
}
