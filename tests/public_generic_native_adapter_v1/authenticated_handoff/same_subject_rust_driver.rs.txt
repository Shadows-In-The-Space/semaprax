// Test-only counter observations; all ownership/provider calls use the actual
// generated safe API. This is not an FFI adapter or a replacement carrier.
#![allow(unsafe_code)]
use @LIBRARY@::{diagnostics, Error, Input, Provider, TRUSTED_BINDING_BYTES, TRUSTED_DESCRIPTOR_BYTES};
extern "C" {
    fn auth_allocations() -> usize;
    fn auth_live() -> usize;
    fn auth_calls() -> usize;
}
fn counts() -> (usize, usize, usize) {
    // SAFETY: these test-only accessors return process-local counters; they
    // accept no pointer or ownership and do not invoke the selected endpoint.
    unsafe { (auth_allocations(), auth_live(), auth_calls()) }
}
fn main() {
    const MODE: u8 = @MODE@;
    const GUARD: bool = @GUARD@;
    let mut malformed = TRUSTED_DESCRIPTOR_BYTES.to_vec();
    malformed[0] ^= 1;
    assert!(matches!(Provider::open(&malformed, TRUSTED_BINDING_BYTES), Err(Error::DescriptorRejected(_))));
    assert_eq!(counts(), (0, 0, 0));
    let mut provider = Provider::open(TRUSTED_DESCRIPTOR_BYTES, TRUSTED_BINDING_BYTES).unwrap();
    for iteration in 0..2 {
        let before = counts().0;
        let result = provider.transform(Input {
            @INPUT0@: vec![1, 7, 13], @INPUT1@: vec![2, 11, 17, 23],
        });
        if MODE == 1 {
            if GUARD {
                let value = result.unwrap();
                assert_eq!(value.@OUTPUT0@, [1, 7, 13]);
                assert_eq!(value.@OUTPUT1@, [2, 11, 17, 23]);
                drop(value.@OUTPUT1@); drop(value.@OUTPUT0@);
            } else { assert!(matches!(result, Err(Error::ExecutionFailed(11)))); }
            assert_eq!(counts().2, iteration + 1);
        } else {
            assert!(matches!(result, Err(Error::CarrierRejected(_))));
            assert_eq!(counts().0, before);
            assert_eq!(counts().2, 0);
        }
        assert_eq!(provider.last_release_status(), 0);
        assert_eq!(diagnostics::live_handles(&provider), 0);
    }
    provider.close().unwrap();
    provider.close().unwrap();
    drop(provider);
    assert_eq!(counts().1, 0);
    println!("rust-authenticated-caller-settled");
}
