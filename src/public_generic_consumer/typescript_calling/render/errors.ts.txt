/** Generated, closed caller error vocabulary. Messages never determine status.
 * Provider execution and cleanup statuses originate in the compiled module.
 */
export type DescriptorReason = "descriptor-bound" | "descriptor-framing" | "descriptor-replay";
export type ProviderReason = "binding-replay" | "binding-bound" | "module-bytes-required"
  | "module-buffer" | "module-bound" | "module-digest" | "module-compile"
  | "module-imports" | "module-exports" | "module-memory" | "module-abi"
  | "module-instance";
export type CarrierReason = "input-shape" | "input-field" | "input-bytes"
  | "provider-closed" | "handle-invalid" | "handle-provider"
  | "handle-kind" | "state-invalid"
  | "carrier-framing" | "carrier-count" | "carrier-trailing";
export type CapacityReason = "leaf-count" | "leaf-bytes" | "total-payload"
  | "carrier-bytes" | "trace-events" | "memory-growth" | "generation" | "host-allocation";
/** Incoming result frames use their own closed refusal class. A hostile
 * provider claim of an over-bound carrier is not a caller-requested capacity
 * failure, even though the bound vocabulary is shared with input admission. */
export type ResultReason = CarrierReason | "leaf-bytes" | "total-payload" | "carrier-bytes";

export type SemapraxPublicGenericError =
  | { readonly kind: "descriptor-rejected"; readonly reason: DescriptorReason }
  | { readonly kind: "provider-mismatch"; readonly reason: ProviderReason }
  | { readonly kind: "carrier-rejected"; readonly reason: CarrierReason }
  | { readonly kind: "capacity-exceeded"; readonly reason: CapacityReason }
  | { readonly kind: "execution-failed"; readonly status: number }
  | { readonly kind: "result-rejected"; readonly reason: ResultReason }
  | { readonly kind: "release-failed"; readonly status: number };

export class SemapraxPublicGenericException extends Error {
  readonly detail: SemapraxPublicGenericError;
  readonly #cleanup: number[] = [];

  constructor(detail: SemapraxPublicGenericError) {
    super("reason" in detail ? `${detail.kind}: ${detail.reason}` : `${detail.kind}: ${detail.status}`);
    this.name = "SemapraxPublicGenericException";
    this.detail = Object.freeze({ ...detail });
  }

  get secondaryCleanupStatuses(): readonly number[] {
    return Object.freeze(this.#cleanup.slice());
  }

  /** Adds evidence only. Never changes the primary error or performs cleanup. */
  recordSecondaryCleanup(status: number): void {
    if (this.#cleanup.length < 2) this.#cleanup.push(status);
  }
}

export function secondaryReleaseNote(status: number): string {
  return `generated-typescript-wasm-consumer: secondary cleanup status ${status}; primary unchanged`;
}
