
// Public byte arrays are informational copies, never mutable replay authority.
const EXPECTED_DESCRIPTOR_BYTES = TRUSTED_DESCRIPTOR_BYTES.slice();
const EXPECTED_BINDING_BYTES = TRUSTED_BINDING_BYTES.slice();
export const MAX_MODULE_BYTES = 16777216;

export function trustedDescriptorAndBinding(): readonly [Uint8Array, Uint8Array] {
  return [EXPECTED_DESCRIPTOR_BYTES.slice(), EXPECTED_BINDING_BYTES.slice()];
}

function parseDescriptorV1(bytes: Uint8Array): readonly Uint8Array[] | null {
  if (bytes.length > 131_072) return null;
  // ignoreBOM=true retains a leading U+FEFF. The reference UTF-8 decoder
  // preserves it too, so a BOM-prefixed schema must never become valid here.
  const decoder = new TextDecoder("utf-8", { fatal: true, ignoreBOM: true });
  let offset = 0;
  const fields: Uint8Array[] = [];
  for (let field = 0; field < 12; field += 1) {
    if (offset > bytes.length || bytes.length - offset < 8) return null;
    let width = 0n;
    for (let index = 0; index < 8; index += 1) {
      width |= BigInt(bytes[offset + index] as number) << BigInt(index * 8);
    }
    if (width > 65_536n) return null;
    offset += 8;
    const length = Number(width);
    if (length > bytes.length - offset) return null;
    let text: string;
    try {
      text = decoder.decode(bytes.subarray(offset, offset + length));
    } catch {
      return null;
    }
    const version = [
      "semaprax.public-generic-descriptor.v1",
      "semaprax.public-generic-boundary-profile.v1",
      "semaprax.public-generic-type-grammar.v1",
    ][field];
    if (version !== undefined && text !== version) return null;
    fields.push(bytes.subarray(offset, offset + length));
    offset += length;
  }
  return offset === bytes.length ? fields : null;
}

function descriptorFieldsReplay(candidateBytes: Uint8Array, trustedBytes: Uint8Array): boolean {
  const candidate = parseDescriptorV1(candidateBytes);
  const trusted = parseDescriptorV1(trustedBytes);
  if (candidate === null || trusted === null) return false;
  return candidate.every((field, index) => bytesEqual(field, trusted[index] as Uint8Array));
}

export function verifyDescriptorAndBinding(
  descriptorBytes: Uint8Array,
  bindingBytes: Uint8Array,
): void {
  let descriptor: Uint8Array;
  let binding: Uint8Array;
  try { descriptor = byteView(descriptorBytes); }
  catch { throw new SemapraxPublicGenericException({ kind: "descriptor-rejected", reason: "descriptor-framing" }); }
  if (descriptor.length > 131072) throw new SemapraxPublicGenericException({ kind: "descriptor-rejected", reason: "descriptor-bound" });
  if (parseDescriptorV1(descriptor) === null) throw new SemapraxPublicGenericException({ kind: "descriptor-rejected", reason: "descriptor-framing" });
  if (!descriptorFieldsReplay(descriptor, EXPECTED_DESCRIPTOR_BYTES)) throw new SemapraxPublicGenericException({ kind: "descriptor-rejected", reason: "descriptor-replay" });
  try { binding = byteView(bindingBytes); }
  catch { throw new SemapraxPublicGenericException({ kind: "provider-mismatch", reason: "binding-replay" }); }
  if (binding.length > 262144) throw new SemapraxPublicGenericException({ kind: "provider-mismatch", reason: "binding-bound" });
  if (!bytesEqual(binding, EXPECTED_BINDING_BYTES)) throw new SemapraxPublicGenericException({ kind: "provider-mismatch", reason: "binding-replay" });
}

/**
 * Independently replay the provider's normalized artifact identity. The
 * binding's fixed-width 71-byte artifact digest occurs exactly once in the
 * authenticated module and is zeroed before domain-separated SHA-256, which
 * breaks the binding/artifact self-reference without trusting a raw offset.
 */
export async function computeModuleArtifactDigest(
  wasmBytes: Uint8Array,
): Promise<string> {
  wasmBytes = byteView(wasmBytes);
  if (wasmBytes.length > MAX_MODULE_BYTES) throw new SemapraxPublicGenericException({ kind: "provider-mismatch", reason: "module-bound" });
  const normalized = wasmBytes.slice();
  const slot = new TextEncoder().encode(TRUSTED_PROVIDER_ARTIFACT_DIGEST);
  if (slot.length !== 71) throw new SemapraxPublicGenericException({ kind: "provider-mismatch", reason: "module-digest" });
  let found = -1;
  for (let offset = 0; offset + slot.length <= normalized.length; offset += 1) {
    let equal = true;
    for (let index = 0; index < slot.length; index += 1) {
      if (normalized[offset + index] !== slot[index]) { equal = false; break; }
    }
    if (!equal) continue;
    if (found !== -1) throw new SemapraxPublicGenericException({ kind: "provider-mismatch", reason: "module-digest" });
    found = offset;
  }
  if (found === -1) throw new SemapraxPublicGenericException({ kind: "provider-mismatch", reason: "module-digest" });
  normalized.fill(0, found, found + slot.length);
  const preimage = new Uint8Array(
    MODULE_ARTIFACT_DIGEST_DOMAIN.length + 8 + normalized.length,
  );
  preimage.set(MODULE_ARTIFACT_DIGEST_DOMAIN, 0);
  new DataView(preimage.buffer).setBigUint64(
    MODULE_ARTIFACT_DIGEST_DOMAIN.length,
    BigInt(normalized.length),
    true,
  );
  preimage.set(normalized, MODULE_ARTIFACT_DIGEST_DOMAIN.length + 8);
  const digestBytes = new Uint8Array(
    await crypto.subtle.digest("SHA-256", preimage),
  );
  let hex = "";
  for (const byte of digestBytes) {
    hex += byte.toString(16).padStart(2, "0");
  }
  return `sha256:${hex}`;
}

export async function verifyModuleArtifactDigest(
  wasmBytes: Uint8Array,
): Promise<void> {
  const computed = await computeModuleArtifactDigest(wasmBytes);
  if (computed !== TRUSTED_PROVIDER_ARTIFACT_DIGEST) {
    throw new SemapraxPublicGenericException({
      kind: "provider-mismatch",
      reason: "module-digest",
    });
  }
}
