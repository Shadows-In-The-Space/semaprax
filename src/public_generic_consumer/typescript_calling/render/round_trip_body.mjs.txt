function inputForLengths(lengths) {
  const input = {};
  const names = Object.keys(sampleInput());
  for (let index = 0; index < names.length; index += 1) {
    input[names[index]] = new Uint8Array(lengths[index] ?? 0).fill(index + 1);
  }
  return input;
}

let failed = 0;
let passed = 0;
async function test(name, run) {
  try { await run(); passed += 1; console.log(`ok - ${name}`); }
  catch (error) { failed += 1; console.error(`FAIL - ${name}`); console.error(error); }
}
async function rejects(operation, kind) {
  await assert.rejects(operation, error => {
    assert.ok(error instanceof SemapraxPublicGenericException);
    assert.equal(error.detail.kind, kind);
    return true;
  });
}

await test("compiled provider executes the checked endpoint and returns independent copies", async () => {
  const provider = await Provider.open(wasmBytes);
  const input = sampleInput();
  const output = provider.transform(input);
  assertOutputShape(output);
  const expected = Object.fromEntries(Object.entries(output).map(([name, value]) => [name, Uint8Array.from(value)]));
  for (const value of Object.values(input)) value.fill(0xff);
  for (const [name, value] of Object.entries(output)) assert.deepEqual(value, expected[name]);
  provider.close();
});

await test("the module-owned provider can execute successive calls", async () => {
  const provider = await Provider.open(wasmBytes);
  for (let count = 0; count < 3; count += 1) {
    const input = sampleInput();
    assertOutputShape(provider.transform(input));
  }
  provider.close();
});

await test("empty owned byte leaves survive the compiled provider lifecycle", async () => {
  const provider = await Provider.open(wasmBytes);
  const input = inputForLengths(Array.from({ length: FIELD_COUNT }, () => 0));
  const output = provider.transform(input);
  assertOutputShape(output);
  provider.close();
});

await test("the per-leaf byte bound is admitted", async () => {
  const provider = await Provider.open(wasmBytes);
  const lengths = Array.from({ length: FIELD_COUNT }, () => 0);
  lengths[0] = MAX_BYTES_PER_LEAF;
  const input = inputForLengths(lengths);
  assertOutputShape(provider.transform(input));
  provider.close();
});

await test("a leaf above the frozen byte bound is refused before provider execution", async () => {
  const provider = await Provider.open(wasmBytes);
  const lengths = Array.from({ length: FIELD_COUNT }, () => 0);
  lengths[0] = MAX_BYTES_PER_LEAF + 1;
  await assert.rejects(() => provider.transform(inputForLengths(lengths)), error => {
    assert.ok(error instanceof SemapraxPublicGenericException);
    assert.equal(error.detail.kind, "capacity-exceeded");
    return true;
  });
  const input = sampleInput();
  assertOutputShape(provider.transform(input));
  provider.close();
});

await test("the exact aggregate payload bound is admitted when the descriptor has 256 leaves", async () => {
  const provider = await Provider.open(wasmBytes);
  const lengths = Array.from({ length: FIELD_COUNT }, () => 0);
  if (FIELD_COUNT === 256) lengths.fill(MAX_BYTES_PER_LEAF);
  const input = inputForLengths(lengths);
  assertOutputShape(provider.transform(input));
  if (FIELD_COUNT === 256) assert.equal(MAX_TOTAL_PAYLOAD_BYTES, lengths.reduce((a, b) => a + b, 0));
  provider.close();
});

await test("an input with an extra property is rejected and the provider remains usable", async () => {
  const provider = await Provider.open(wasmBytes);
  const malformed = { ...sampleInput(), unexpected: new Uint8Array(0) };
  await assert.rejects(() => provider.transform(malformed), error => {
    assert.ok(error instanceof SemapraxPublicGenericException);
    assert.equal(error.detail.kind, "carrier-rejected");
    return true;
  });
  const input = sampleInput();
  assertOutputShape(provider.transform(input));
  provider.close();
});

await test("a modified trusted descriptor is refused before module compilation", async () => {
  const mutated = TRUSTED_DESCRIPTOR_BYTES.slice();
  mutated[0] = mutated[0] ^ 0xff;
  await rejects(() => Provider.open(wasmBytes, { descriptorBytes: mutated }), "descriptor-rejected");
});

await test("a modified trusted binding is refused before module compilation", async () => {
  const mutated = TRUSTED_BINDING_BYTES.slice();
  mutated[mutated.length - 1] = mutated[mutated.length - 1] ^ 0xff;
  await rejects(() => Provider.open(wasmBytes, { bindingBytes: mutated }), "provider-mismatch");
});

await test("a changed descriptor document is refused", async () => {
  const mutated = new Uint8Array(TRUSTED_DESCRIPTOR_BYTES.length + 8);
  mutated.set(TRUSTED_DESCRIPTOR_BYTES);
  mutated.set(new TextEncoder().encode("EXTENDED"), TRUSTED_DESCRIPTOR_BYTES.length);
  await rejects(() => Provider.open(wasmBytes, { descriptorBytes: mutated }), "descriptor-rejected");
});

await test("a module digest mismatch is refused before instantiation", async () => {
  const mutated = Uint8Array.from(wasmBytes);
  mutated[mutated.length - 1] = mutated[mutated.length - 1] ^ 0xff;
  const original = WebAssembly.instantiate;
  let instantiated = 0;
  WebAssembly.instantiate = function (...args) { instantiated += 1; return Reflect.apply(original, WebAssembly, args); };
  try { await rejects(() => Provider.open(mutated), "provider-mismatch"); }
  finally { WebAssembly.instantiate = original; }
  assert.equal(instantiated, 0);
});

await test("a precompiled module is refused because its original bytes are unavailable", async () => {
  const module = await WebAssembly.compile(wasmBytes);
  await rejects(() => Provider.open(module), "provider-mismatch");
});

await test("module bytes are snapshotted before asynchronous digest and compilation", async () => {
  const mutable = Uint8Array.from(wasmBytes);
  const opening = Provider.open(mutable);
  mutable.fill(0);
  const provider = await opening;
  const input = sampleInput();
  assertOutputShape(provider.transform(input));
  provider.close();
});

await test("close is idempotent and later calls are refused", async () => {
  const provider = await Provider.open(wasmBytes);
  provider.close();
  provider.close();
  assert.throws(() => provider.transform(sampleInput()), error => {
    assert.ok(error instanceof SemapraxPublicGenericException);
    assert.equal(error.detail.kind, "carrier-rejected");
    return true;
  });
});

await test("independent providers keep module-issued handles isolated", async () => {
  const first = await Provider.open(wasmBytes);
  const second = await Provider.open(wasmBytes);
  const input = sampleInput();
  assertOutputShape(first.transform(input));
  assertOutputShape(second.transform(input));
  first.close();
  second.close();
});
