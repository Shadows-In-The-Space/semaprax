use std::{ffi::c_void, ptr};
unsafe extern "C" {
    fn auth_allocations() -> usize;
    fn auth_live() -> usize;
    fn auth_calls() -> usize;
    fn spx_pg_provider_open_v1(d: *const u8, dn: usize, b: *const u8, bn: usize, p: *mut *mut c_void) -> i32;
    fn spx_pg_authenticated_generation_v1(p: *mut c_void, generation: *mut u64) -> i32;
    fn spx_pg_authenticated_input_prepare_v1(p: *mut c_void, generation: u64, owner: u32,
        cleanup: *const u8, cleanup_len: usize, frame: *const u8, frame_len: usize, input: *mut *mut c_void) -> i32;
    fn spx_pg_input_prepare_v1(p: *mut c_void, frame: *const u8, len: usize, input: *mut *mut c_void) -> i32;
    fn spx_pg_call_v1(p: *mut c_void, input: *mut c_void, result: *mut *mut c_void) -> i32;
    fn spx_pg_result_export_v1(result: *mut c_void, out: *mut u8, len: usize, required: *mut usize) -> i32;
    fn spx_pg_result_release_v1(result: *mut *mut c_void) -> i32;
    fn spx_pg_provider_close_v1(provider: *mut *mut c_void) -> i32;
}
unsafe fn refused(p: *mut c_void, generation: u64, owner: u32, plan: &[u8], frame: &[u8], expected: i32) {
    let before=(auth_allocations(),auth_live(),auth_calls());
    let mut input=ptr::null_mut();
    assert_eq!(spx_pg_authenticated_input_prepare_v1(p,generation,owner,plan.as_ptr(),plan.len(),frame.as_ptr(),frame.len(),&mut input),expected);
    assert!(input.is_null());
    assert_eq!(before,(auth_allocations(),auth_live(),auth_calls()));
}
fn main() { unsafe {
    let mut provider=ptr::null_mut();
    assert_eq!(spx_pg_provider_open_v1(DESCRIPTOR.as_ptr(),DESCRIPTOR.len(),BINDING.as_ptr(),BINDING.len(),&mut provider),0);
    let mut generation=0;
    assert_eq!(spx_pg_authenticated_generation_v1(provider,&mut generation),0);
    refused(provider,generation+1,0,CLEANUP,CANONICAL,8);
    refused(provider,generation,1,CLEANUP,CANONICAL,7);
    refused(provider,generation,0,DESCRIPTOR,CANONICAL,14);
    refused(provider,generation,0,CLEANUP,WRONG_PATH,14);
    refused(provider,generation,0,CLEANUP,UNKNOWN_DIRECTION,5);
    refused(provider,generation,0,CLEANUP,DUPLICATE_PATH,5);
    refused(provider,generation,0,CLEANUP,INVALID_UTF8,5);
    refused(provider,generation,0,CLEANUP,OVERSIZED_METADATA,5);
    refused(provider,generation,0,CLEANUP,WRONG_TAG,5);
    refused(provider,generation,0,CLEANUP,CORRUPT,14);
    refused(provider,generation,0,CLEANUP,OVERSIZED,6);
    for n in 0..CANONICAL.len() { refused(provider,generation,0,CLEANUP,&CANONICAL[..n],5); }
    let flat=[2,0,0,0,0,0,0,0,3,0,0,0,0,0,0,0,1,7,13,3,0,0,0,0,0,0,0,2,7,13];
    refused(provider,generation,0,CLEANUP,&flat,5);
    let before=auth_allocations();
    let mut input=ptr::null_mut();
    assert_eq!(spx_pg_input_prepare_v1(provider,flat.as_ptr(),flat.len(),&mut input),5);
    assert!(input.is_null()); assert_eq!(auth_allocations(),before);
    assert_eq!(spx_pg_authenticated_input_prepare_v1(provider,generation,0,CLEANUP.as_ptr(),CLEANUP.len(),CANONICAL.as_ptr(),CANONICAL.len(),&mut input),0);
    assert_eq!(auth_calls(),0);
    let mut result=ptr::null_mut();
    assert_eq!(spx_pg_call_v1(provider,input,&mut result),EXPECT_CALL_STATUS);
    assert_eq!(auth_calls(),1);
    let mut duplicate=ptr::null_mut();
    assert_eq!(spx_pg_call_v1(provider,input,&mut duplicate),8);
    assert_eq!(auth_calls(),1);
    if EXPECT_CALL_STATUS==0 {
        let mut out=[0;30]; let mut required=0;
        assert_eq!(spx_pg_result_export_v1(result,out.as_mut_ptr(),out.len(),&mut required),0);
        assert_eq!(required,flat.len()); assert_eq!(out,flat);
        assert_eq!(spx_pg_result_release_v1(&mut result),0);
    }
    assert!(result.is_null());
    refused(provider,generation,0,CLEANUP,CANONICAL,8);
    assert_eq!(spx_pg_provider_close_v1(&mut provider),0);
    assert_eq!(auth_live(),0);
    println!("authenticated-native-handoff-settled");
} }
