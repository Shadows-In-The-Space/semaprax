/** Generated TypeScript client for the frozen module-owned public-generic
 * Wasm provider ABI. The compiled module owns provider state, allocations,
 * handles, call settlement, and physical release. This client only validates
 * the artifact, marshals the public carrier through the reserved scratch
 * region, and guarantees that every acquired opaque value is released.
 */
import { MAX_CARRIER_BYTES, decodeOutput, encodeInput } from "./carrier.js";
import { MAX_MODULE_BYTES, trustedDescriptorAndBinding,
  verifyDescriptorAndBinding, verifyModuleArtifactDigest } from "./descriptor.js";
import { SemapraxPublicGenericException } from "./errors.js";
import type { CarrierReason, ProviderReason, SemapraxPublicGenericError } from "./errors.js";
import type { Input, Output } from "./types.js";

const PAGE_BYTES = 65536;
// The frozen emitter reserves a public scratch arena and an equal private
// provider arena after its fixed prefix.
const SCRATCH_BASE_BYTES = 393216;
const MAX_MEMORY_BYTES = SCRATCH_BASE_BYTES + 2 * MAX_CARRIER_BYTES;
const STATUS_OK = 0;
const STATUS_BUFFER_TOO_SMALL = 12;
const ABI_EXPORTS = [
  ["memory", "memory"],
  ["spx_pg_v1_scratch_ptr", "function"],
  ["spx_pg_v1_scratch_reserve", "function"],
  ["spx_pg_v1_scratch_capacity", "function"],
  ["spx_pg_v1_open", "function"],
  ["spx_pg_v1_input_prepare", "function"],
  ["spx_pg_v1_call", "function"],
  ["spx_pg_v1_result_export", "function"],
  ["spx_pg_v1_value_release", "function"],
  ["spx_pg_v1_result_release", "function"],
  ["spx_pg_v1_provider_close", "function"],
] as const;

type I32Fn = (...args: number[]) => number;
type LaneFn = (...args: number[]) => bigint;
interface ProviderAbi {
  readonly scratchPtr: I32Fn;
  readonly scratchReserve: LaneFn;
  readonly scratchCapacity: I32Fn;
  readonly open: LaneFn;
  readonly inputPrepare: LaneFn;
  readonly call: LaneFn;
  readonly resultExport: LaneFn;
  readonly valueRelease: I32Fn;
  readonly resultRelease: I32Fn;
  readonly providerClose: I32Fn;
}
interface Lane { readonly status: number; readonly value: number; }

function failure(detail: SemapraxPublicGenericError): SemapraxPublicGenericException {
  return new SemapraxPublicGenericException(detail);
}
function carrier(reason: CarrierReason): SemapraxPublicGenericException {
  return failure({ kind: "carrier-rejected", reason });
}
function mismatch(reason: ProviderReason): SemapraxPublicGenericException {
  return failure({ kind: "provider-mismatch", reason });
}
function normalized(error: unknown): SemapraxPublicGenericException {
  return error instanceof SemapraxPublicGenericException
    ? error
    : failure({ kind: "execution-failed", status: 11 });
}
function readLane(raw: bigint): Lane {
  const bits = BigInt.asUintN(64, raw);
  return {
    status: Number(bits & 0xffff_ffffn),
    value: Number((bits >> 32n) & 0xffff_ffffn),
  };
}
function statusFailure(status: number, phase: "open" | "prepare" | "call" | "export"): SemapraxPublicGenericException {
  switch (status) {
    case 1: return failure({ kind: "descriptor-rejected", reason: "descriptor-framing" });
    case 2: return failure({ kind: "descriptor-rejected", reason: "descriptor-replay" });
    case 3:
    case 4: return mismatch("binding-replay");
    case 5: return phase === "export"
      ? failure({ kind: "result-rejected", reason: "carrier-framing" })
      : carrier("carrier-framing");
    case 6: return failure({ kind: "capacity-exceeded", reason: "carrier-bytes" });
    case 7: return carrier("state-invalid");
    case 8: return carrier("handle-invalid");
    case 12: return failure({ kind: "capacity-exceeded", reason: "carrier-bytes" });
    default: return failure({ kind: "execution-failed", status });
  }
}
function releaseFailure(status: number): SemapraxPublicGenericException {
  return failure({ kind: "release-failed", status });
}
function snapshotModuleBytes(value: BufferSource): Uint8Array {
  try {
    if (value instanceof ArrayBuffer) {
      const length = value.byteLength;
      if (length > MAX_MODULE_BYTES
        || ((value as ArrayBuffer & { readonly resizable?: boolean }).resizable ?? false)) {
        throw mismatch("module-buffer");
      }
      return new Uint8Array(value.slice(0));
    }
    if (!ArrayBuffer.isView(value) || !(value.buffer instanceof ArrayBuffer)
      || ((value.buffer as ArrayBuffer & { readonly resizable?: boolean }).resizable ?? false)) {
      throw mismatch("module-buffer");
    }
    const view = new Uint8Array(value.buffer, value.byteOffset, value.byteLength);
    if (view.length > MAX_MODULE_BYTES) throw mismatch("module-bound");
    return Uint8Array.prototype.slice.call(view) as Uint8Array;
  } catch (error) {
    if (error instanceof SemapraxPublicGenericException) throw error;
    throw mismatch("module-buffer");
  }
}
function nonnegativeI32(value: number): boolean {
  return Number.isInteger(value) && value >= 0 && value <= 0xffff_ffff;
}
function exactAbi(module: WebAssembly.Module): void {
  const actual = WebAssembly.Module.exports(module)
    .map(({ name, kind }) => `${name}\0${kind}`)
    .sort();
  const expected = ABI_EXPORTS.map(([name, kind]) => `${name}\0${kind}`).sort();
  if (actual.length !== expected.length || actual.some((value, index) => value !== expected[index])) {
    throw mismatch("module-exports");
  }
  if (WebAssembly.Module.imports(module).length !== 0) throw mismatch("module-imports");
}

/** An opaque JS wrapper around a module-issued ID. It does not track ID
 * validity or ownership; all such decisions remain inside the Wasm module.
 */
class OpaqueValue {
  readonly #providerTag: object;
  readonly #id: number;
  readonly #kind: "input" | "result";
  constructor(providerTag: object, id: number, kind: "input" | "result") {
    this.#providerTag = providerTag;
    this.#id = id;
    this.#kind = kind;
  }
  static read(value: unknown): readonly [object, number, "input" | "result"] {
    if (value === null || typeof value !== "object" || !(#providerTag in value)) {
      throw carrier("handle-invalid");
    }
    return [value.#providerTag, value.#id, value.#kind];
  }
}

export class Provider {
  readonly #memory: WebAssembly.Memory;
  readonly #abi: ProviderAbi;
  readonly #tag = {};
  readonly #providerId: number;
  readonly #scratchPtr: number;
  readonly #scratchCapacity: number;
  #closed = false;

  private constructor(memory: WebAssembly.Memory, abi: ProviderAbi, providerId: number,
    scratchPtr: number, scratchCapacity: number) {
    this.#memory = memory;
    this.#abi = abi;
    this.#providerId = providerId;
    this.#scratchPtr = scratchPtr;
    this.#scratchCapacity = scratchCapacity;
  }

  static async open(moduleOrBytes: WebAssembly.Module | BufferSource,
    options: { descriptorBytes?: Uint8Array; bindingBytes?: Uint8Array } = {}): Promise<Provider> {
    const [descriptor, binding] = trustedDescriptorAndBinding();
    verifyDescriptorAndBinding(options.descriptorBytes ?? descriptor, options.bindingBytes ?? binding);
    if (moduleOrBytes instanceof WebAssembly.Module) throw mismatch("module-bytes-required");
    // Snapshot before the first await so authenticated bytes and compiled bytes
    // cannot diverge through a caller mutation.
    const bytes = snapshotModuleBytes(moduleOrBytes);
    await verifyModuleArtifactDigest(bytes);
    let module: WebAssembly.Module;
    try { module = await WebAssembly.compile(bytes); }
    catch { throw mismatch("module-compile"); }
    exactAbi(module);
    let instance: WebAssembly.Instance;
    try { instance = await WebAssembly.instantiate(module, {}); }
    catch { throw mismatch("module-instance"); }

    const exports = instance.exports;
    const memory = exports.memory;
    if (!(memory instanceof WebAssembly.Memory) || !(memory.buffer instanceof ArrayBuffer)
      || memory.buffer.byteLength < PAGE_BYTES || memory.buffer.byteLength > MAX_MEMORY_BYTES) {
      throw mismatch("module-memory");
    }
    const fn = (name: string): I32Fn | LaneFn => {
      const value = exports[name];
      if (typeof value !== "function") throw mismatch("module-abi");
      return value as I32Fn | LaneFn;
    };
    const abi: ProviderAbi = {
      scratchPtr: fn("spx_pg_v1_scratch_ptr") as I32Fn,
      scratchReserve: fn("spx_pg_v1_scratch_reserve") as LaneFn,
      scratchCapacity: fn("spx_pg_v1_scratch_capacity") as I32Fn,
      open: fn("spx_pg_v1_open") as LaneFn,
      inputPrepare: fn("spx_pg_v1_input_prepare") as LaneFn,
      call: fn("spx_pg_v1_call") as LaneFn,
      resultExport: fn("spx_pg_v1_result_export") as LaneFn,
      valueRelease: fn("spx_pg_v1_value_release") as I32Fn,
      resultRelease: fn("spx_pg_v1_result_release") as I32Fn,
      providerClose: fn("spx_pg_v1_provider_close") as I32Fn,
    };

    // Reserve the frozen maximum carrier once, before open creates a live
    // provider. Subsequent calls reuse only this public scratch span.
    const required = MAX_CARRIER_BYTES;
    const reserved = readLane(abi.scratchReserve(required));
    if (reserved.status !== STATUS_OK) throw statusFailure(reserved.status, "open");
    const scratchPtr = abi.scratchPtr();
    const scratchCapacity = abi.scratchCapacity();
    if (!nonnegativeI32(scratchPtr) || !nonnegativeI32(scratchCapacity)
      || reserved.value !== scratchPtr || scratchCapacity < required
      || memory.buffer.byteLength > MAX_MEMORY_BYTES
      || scratchPtr + scratchCapacity > memory.buffer.byteLength) throw mismatch("module-memory");

    const descriptorPtr = scratchPtr;
    const bindingPtr = descriptorPtr + descriptor.length;
    if (bindingPtr + binding.length > scratchPtr + scratchCapacity) throw mismatch("module-memory");
    let view = new Uint8Array(memory.buffer);
    view.set(descriptor, descriptorPtr);
    view.set(binding, bindingPtr);
    // The open export independently replays the exact descriptor and binding.
    const opened = readLane(abi.open(descriptorPtr, descriptor.length, bindingPtr, binding.length));
    if (opened.status !== STATUS_OK || opened.value === 0) {
      const primary = opened.status === STATUS_OK
        ? mismatch("module-abi")
        : statusFailure(opened.status, "open");
      if (opened.value !== 0) {
        const cleanup = abi.providerClose(opened.value);
        if (cleanup !== STATUS_OK) primary.recordSecondaryCleanup(cleanup);
      }
      throw primary;
    }
    return new Provider(memory, abi, opened.value, scratchPtr, scratchCapacity);
  }

  #live(): void {
    if (this.#closed) throw carrier("provider-closed");
  }
  #memoryView(): Uint8Array {
    if (!(this.#memory.buffer instanceof ArrayBuffer)
      || this.#memory.buffer.byteLength > MAX_MEMORY_BYTES) throw mismatch("module-memory");
    const view = new Uint8Array(this.#memory.buffer);
    if (this.#scratchPtr + this.#scratchCapacity > view.length) throw mismatch("module-memory");
    return view;
  }
  #copyIntoScratch(bytes: Uint8Array): void {
    if (bytes.length > this.#scratchCapacity) {
      throw failure({ kind: "capacity-exceeded", reason: "carrier-bytes" });
    }
    // Reacquire after every module call: Wasm memory growth detaches old views.
    this.#memoryView().set(bytes, this.#scratchPtr);
  }
  #requireValue(value: unknown, kind: "input" | "result"): number {
    this.#live();
    const [providerTag, id, actualKind] = OpaqueValue.read(value);
    if (providerTag !== this.#tag) throw carrier("handle-provider");
    if (actualKind !== kind) throw carrier("handle-kind");
    return id;
  }
  #prepareInput(input: Input): OpaqueValue {
    this.#live();
    const carrierBytes = encodeInput(input);
    this.#copyIntoScratch(carrierBytes);
    const prepared = readLane(this.#abi.inputPrepare(this.#providerId, this.#scratchPtr, carrierBytes.length));
    if (prepared.status !== STATUS_OK) {
      const primary = statusFailure(prepared.status, "prepare");
      if (prepared.value !== 0) {
        const cleanup = this.#abi.valueRelease(prepared.value);
        if (cleanup !== STATUS_OK) primary.recordSecondaryCleanup(cleanup);
      }
      throw primary;
    }
    if (prepared.value === 0) throw mismatch("module-abi");
    return new OpaqueValue(this.#tag, prepared.value, "input");
  }
  #call(input: OpaqueValue): OpaqueValue {
    const inputId = this.#requireValue(input, "input");
    const result = readLane(this.#abi.call(this.#providerId, inputId));
    if (result.status !== STATUS_OK) {
      const primary = statusFailure(result.status, "call");
      if (result.value !== 0) {
        const cleanup = this.#abi.resultRelease(result.value);
        if (cleanup !== STATUS_OK) primary.recordSecondaryCleanup(cleanup);
      }
      throw primary;
    }
    if (result.value === 0) throw mismatch("module-abi");
    return new OpaqueValue(this.#tag, result.value, "result");
  }
  #exportResult(result: OpaqueValue): Output {
    const resultId = this.#requireValue(result, "result");
    const probe = readLane(this.#abi.resultExport(resultId, this.#scratchPtr, 0));
    if (probe.status !== STATUS_BUFFER_TOO_SMALL || probe.value === 0
      || probe.value > MAX_CARRIER_BYTES || probe.value > this.#scratchCapacity) {
      throw probe.status === STATUS_BUFFER_TOO_SMALL
        ? failure({ kind: "capacity-exceeded", reason: "carrier-bytes" })
        : statusFailure(probe.status, "export");
    }
    const copied = readLane(this.#abi.resultExport(resultId, this.#scratchPtr, probe.value));
    if (copied.status !== STATUS_OK || copied.value !== probe.value) {
      throw copied.status === STATUS_OK ? mismatch("module-abi") : statusFailure(copied.status, "export");
    }
    // Detach output ownership from Wasm scratch before releasing the module
    // result or reusing scratch for the next request.
    const ownedCarrier = this.#memoryView().slice(this.#scratchPtr, this.#scratchPtr + copied.value);
    return decodeOutput(ownedCarrier);
  }
  #releaseInput(value: OpaqueValue): SemapraxPublicGenericException | null {
    try {
      const id = this.#requireValue(value, "input");
      const status = this.#abi.valueRelease(id);
      return status === STATUS_OK ? null : releaseFailure(status);
    } catch (error) { return normalized(error); }
  }
  #releaseResult(value: OpaqueValue): SemapraxPublicGenericException | null {
    try {
      const id = this.#requireValue(value, "result");
      const status = this.#abi.resultRelease(id);
      return status === STATUS_OK ? null : releaseFailure(status);
    } catch (error) { return normalized(error); }
  }
  #combine(primary: SemapraxPublicGenericException | null,
    cleanup: SemapraxPublicGenericException | null): SemapraxPublicGenericException | null {
    if (!cleanup) return primary;
    if (!primary) return cleanup;
    const status = "status" in cleanup.detail ? cleanup.detail.status : 11;
    primary.recordSecondaryCleanup(status);
    return primary;
  }

  transform(input: Input): Output {
    this.#live();
    let prepared: OpaqueValue | null = null;
    let result: OpaqueValue | null = null;
    let output: Output | null = null;
    let primary: SemapraxPublicGenericException | null = null;
    try {
      prepared = this.#prepareInput(input);
      result = this.#call(prepared);
      // Successful module call consumes the input handle at its transfer
      // boundary; only the result remains for explicit release.
      prepared = null;
      output = this.#exportResult(result);
    } catch (error) { primary = normalized(error); }
    if (result) primary = this.#combine(primary, this.#releaseResult(result));
    if (prepared) primary = this.#combine(primary, this.#releaseInput(prepared));
    if (primary) throw primary;
    if (output === null) throw carrier("state-invalid");
    return output;
  }

  close(): void {
    if (this.#closed) return;
    const status = this.#abi.providerClose(this.#providerId);
    if (status !== STATUS_OK) throw releaseFailure(status);
    this.#closed = true;
  }
}
